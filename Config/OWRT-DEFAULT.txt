# =====================================================================
#  OWRT-DEFAULT.txt —— 显式默认出厂配置（局域网 / Wi-Fi / 系统参数）
# =====================================================================
#
#  【作用】
#  本文件是固件出厂默认参数的唯一来源，集中管理后台地址、系统标识、
#  Wi-Fi 名称 / 密钥 / 加密 / 频宽、国家码、时区等全部默认值。
#  构建时由 WRT-CORE 工作流的「Load Default Settings」步骤读取，并经
#  Scripts/Settings.sh 与 Scripts/Handles.sh 注入固件首次启动脚本。
#
#  【如何自定义】
#  1. 直接修改本文件任意「配置项 = 值」并提交即可，无需改动任何脚本或工作流；
#  2. 工作流 inputs（WRT_IP / WRT_PW / WRT_SSID / WRT_WORD 等）显式传入的值
#     优先于本文件；未显式传入时一律使用本文件默认值；
#  3. 取值请严格遵循各项注释中的可选项与格式要求，非法值可能导致对应功能
#     不可用（例如加密策略写错会导致 Wi-Fi 无法开启）。
#
#  【注意事项】
#  - 文件编码必须为 UTF-8，行尾必须为 LF（CRLF 会导致部分加载环境失败）；
#  - 配置项命名只能包含字母、数字与下划线，值不要加引号；
#  - 以 # 开头的行是注释，不会被加载；删除某行 = 使用脚本内置兜底默认值；
#  - 本文件仅用于「显式默认值」，实际生效以工作流 inputs 优先、本文件次之、
#    脚本兜底最末。
# =====================================================================


# ============================== 后台管理 ==============================
# 后台管理地址（Web 登录 IPv4 地址）
# 格式：合法 IPv4，如 192.168.1.1 / 10.0.0.1
# 该地址同时作为设备 LAN 口 IP 与 DHCP 网段起点
WRT_IP=192.168.10.1

# 后台登录密码提示文本（仅用于 Release 发布说明展示，不影响固件实际密码）
# 固件默认 root 无密码，ImmortalWrt 首次登录时由 LuCI 引导设置
# 留空（WRT_PW=）即保持无密码
WRT_PW=无


# ============================== 系统标识 ==============================
# 系统主机名（LuCI 状态页 / SSH 提示符中显示的名称）
# 格式：仅限字母、数字、下划线、连字符
WRT_NAME=OWRT

# 默认 Web 主题（对应主题需在 Scripts/Packages.sh 中已克隆，且 Scripts/Settings.sh
# 里有配套包映射；未编入的主题会被默认主题种子脚本跳过）
# 可选项：vantage（默认，anothaDev/luci-theme-vantage）/ aurora / argon / material / bootstrap 等
# 注：Aurora 仍会随固件编入（含 luci-app-aurora-config），只是不再默认，
#     可在「系统 → 语言和界面」中随时切换
WRT_THEME=vantage


# ============================== Wi-Fi 无线 ==============================
# 【设备 WiFi 世代区分】
# 默认值按设备 WiFi 世代区分：
#   - AP3000M 为 WiFi6（802.11ax，MT7981B）；
#   - H5000M 为 WiFi7（802.11be，MT7987A + MT7996E + MT5700M）。
#
# 【两组变量说明】
#   - 无后缀变量（WRT_SSID 等）为 WiFi6 世代默认值（保持历史兼容）；
#   - _WIFI7 后缀变量（WRT_SSID_WIFI7 等）为 WiFi7 世代默认值，仅当
#     Scripts/Settings.sh 识别到 WiFi7 设备（H5000M）时启用；未填写时
#     自动回退到对应的 WiFi6 默认值。
#
# 【htmode 前缀与世代的对应关系】
#   - WiFi6（AX）2.4G / 5G 均用 HE（802.11ax）；
#   - WiFi7（BE）2.4G / 5G 均用 EHT（802.11be）；
#   - 同机两个频段世代一致（2.4G 与 5G 同步对齐）。
#
# 【频宽上限】
#   - AP3000M（MT7981）5G 硬件上限为 160MHz（AX），默认即 160MHz；
#   - H5000M 为 2.4G / 5G 双频设备，5G 上限同为 160MHz；
#   - 320MHz 仅限 6GHz 频段（802.11be 特性），本机不启用。

# ---- 无线（WiFi6 默认 · AP3000M 等 802.11ax 设备）----

# Wi-Fi 默认名称（SSID），2.4G 与 5G 统一使用
# 长度 1~32 字符，建议使用字母与数字，避免中文与特殊符号
WRT_SSID=OWRT

# Wi-Fi 默认密钥（密码）
# 长度至少 8 字符（WPA2 / WPA3 均要求）
WRT_WORD=12345678

# Wi-Fi 加密策略（决定认证方式与终端兼容性）
# 可选项：
#   psk-mixed   mixed WPA/WPA2 PSK (CCMP)（默认，兼容新老终端）
#   psk2        WPA2-PSK
#   sae-mixed   WPA3 / WPA2 混合（终端较新时推荐）
#   sae         WPA3-PSK（纯 WPA3，老终端无法连接）
# 注意：构建时脚本会根据驱动栈自动追加必要后缀（如 MTK 闭源栈的 +ccmp），
#       此处只需填写裸策略名，无需手动拼接
WIFI_ENCRYPTION=psk-mixed

# 2.4G 频宽（MHz），WiFi6 下对应 htmode=HE40（AX）
# 可选项：20 / 40（40 吞吐更高，20 抗干扰更强）
WIFI_2G_WIDTH=40

# 5G 频宽（MHz），WiFi6 下对应 htmode=HE160（AX）
# 可选项：20 / 40 / 80 / 160
# 注意：AP3000M（MT7981）5G 硬件上限为 160MHz（AX），默认 160 时不降级
WIFI_5G_WIDTH=160

# ---- 无线（WiFi7 默认 · H5000M 等 802.11be 设备）----
# 仅当 Scripts/Settings.sh 识别到 WiFi7 设备（H5000M）时启用；
# 各项未填写时自动回退到上方对应的 WiFi6 默认值。

# Wi-Fi 默认名称（SSID），2.4G 与 5G 统一使用
WRT_SSID_WIFI7=OWRT

# Wi-Fi 默认密钥（密码）
WRT_WORD_WIFI7=12345678

# Wi-Fi 加密策略（决定认证方式与终端兼容性），可选值同 WiFi6 的 WIFI_ENCRYPTION
WIFI_ENCRYPTION_WIFI7=psk-mixed

# 2.4G 频宽（MHz），WiFi7 下对应 htmode=EHT40（BE，与 5G 世代同步）
WIFI_2G_WIDTH_WIFI7=40

# 5G 频宽（MHz），WiFi7 下对应 htmode=EHT160（BE）
WIFI_5G_WIDTH_WIFI7=160

# 国家 / 地区码（可选项见下方「国家与时区」区块的 WIFI_COUNTRY 说明）
WIFI_COUNTRY_WIFI7=CN


# ============================== 国家与时区 ==============================
# 国家 / 地区码（决定可用信道、最大发射功率与合规频段）
# 常见取值：CN（中国，默认）/ US（美国）/ DE（德国）/ GB（英国）/ SG（新加坡）
# 填错可能导致部分信道不可用或发射功率受限
WIFI_COUNTRY=CN

# 系统时区：POSIX 时区串与 IANA 时区名需成对配置
#   WRT_TIMEZONE   POSIX 格式，如 CST-8（东八区，固定 8 小时偏移的写法）
#   WRT_ZONENAME   IANA 格式，如 Asia/Shanghai（支持夏令时的完整时区名）
# 常用组合：CST-8 + Asia/Shanghai（北京时间）
WRT_TIMEZONE=CST-8
WRT_ZONENAME=Asia/Shanghai
