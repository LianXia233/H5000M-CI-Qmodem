# input-support 是 ImmortalWrt hardware-support 的特性门控包。
# 历史背景：2026-10 曾出现 USB serial/acm 依赖该门控，未启用时整条 QModem
# 依赖链被静默隐藏（2026-10-11 复现）。当前 master 中该依赖已收窄到
# input-evdev / input-joydev / usb-hid 的 evdev 选择，USB serial 不再依赖它，
# 但这里仍显式启用作为兼容保护：QModem 配置若切换到仍保留旧依赖的分支，
# 删除本行会再次导致整个前端与核心包被丢弃。不要删除。
CONFIG_PACKAGE_input-support=y
# QMI 驱动 choice 符号挂在 qmodem 主包上（上游 964d8dd32 定义为 PACKAGE_qmodem_INCLUDE_*；
# 旧名 luci-app-qmodem_INCLUDE_* 不存在，defconfig 会静默丢弃，导致 choice 落到上游默认
# 的 vendor 驱动，kmod-qmi_wwan_q/f/s 三件套进固件，与通用 QMI 产生同名模块冲突）。
# 必须用正确符号显式选通用 QMI（kmod-usb-net-qmi-wwan）；NSS 选项仅 qualcommax 平台可见。
CONFIG_PACKAGE_qmodem_INCLUDE_generic-qmi-wwan=y
CONFIG_PACKAGE_qmodem_INCLUDE_SEAL=y
# 拨号 CM choice：上游默认 Tom 定制版（quectel-CM-5G-M），显式声明防止上游改默认值后静默漂移
CONFIG_PACKAGE_qmodem_USE_TOM_CUSTOMIZED_QUECTEL_CM=y
